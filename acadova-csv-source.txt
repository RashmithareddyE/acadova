const DAYS = ['monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday', 'sunday'];
const DAY_ALIASES = new Map([
  ['monday', 0], ['mon', 0], ['tuesday', 1], ['tue', 1], ['tues', 1],
  ['wednesday', 2], ['wed', 2], ['thursday', 3], ['thu', 3], ['thur', 3], ['thurs', 3],
  ['friday', 4], ['fri', 4], ['saturday', 5], ['sat', 5], ['sunday', 6], ['sun', 6],
]);

function parseCsv(text, delimiter) {
  const rows = []; let row = []; let value = ''; let quoted = false;
  for (let i = 0; i < text.length; i += 1) {
    const char = text[i];
    if (char === '"' && quoted && text[i + 1] === '"') { value += '"'; i += 1; }
    else if (char === '"') quoted = !quoted;
    else if (char === delimiter && !quoted) { row.push(value.trim()); value = ''; }
    else if ((char === '\n' || char === '\r') && !quoted) {
      if (char === '\r' && text[i + 1] === '\n') i += 1;
      row.push(value.trim()); if (row.some(Boolean)) rows.push(row); row = []; value = '';
    } else value += char;
  }
  row.push(value.trim()); if (row.some(Boolean)) rows.push(row);
  return rows;
}

function weekday(value) {
  // Tesseract often reads compact timetable labels as M0N, TUE., or
  // separates letters with whitespace. Normalize only common visual OCR
  // confusions, then require a complete known alias (never just a prefix).
  const key = String(value || '').trim().toLowerCase()
    .replace(/0/g, 'o').replace(/[1|!l]/g, 'i').replace(/5/g, 's')
    .replace(/[^a-z]/g, '');
  return DAY_ALIASES.get(key) ?? null;
}

function hasWeekdayText(text) {
  const tokens = String(text || '').match(/[a-z0-9|!]+/gi) || [];
  for (let index = 0; index < tokens.length; index += 1) {
    if (weekday(tokens[index]) !== null) return true;
    let combined = '';
    for (let offset = index; offset < Math.min(tokens.length, index + 7); offset += 1) {
      if (tokens[offset].length > 2) break;
      combined += tokens[offset];
      if (weekday(combined) !== null && offset > index) return true;
    }
  }
  return false;
}

function normalizeTime(value) {
  const match = String(value || '').trim().replace('.', ':').match(/^(\d{1,2})(?::([0-5]\d))?\s*(am|pm)?$/i);
  if (!match) return null;
  let hour = Number(match[1]); const minute = Number(match[2] || 0); const meridiem = match[3]?.toLowerCase();
  if (meridiem) {
    if (hour < 1 || hour > 12) return null;
    if (hour === 12) hour = 0;
    if (meridiem === 'pm') hour += 12;
  } else if (hour > 23) return null;
  return `${String(hour).padStart(2, '0')}:${String(minute).padStart(2, '0')}`;
}

function rowResult({ subjectName = '', subjectCode = '', day, start, end, location = '', teacher = '', sessionType = '', section = '' }, sourceLine) {
  const normalizedDay = Number.isInteger(day) ? day : weekday(day);
  const startTime = normalizeTime(start); const endTime = normalizeTime(end);
  const issues = [];
  if (!subjectName.trim()) issues.push('Subject is missing.');
  if (normalizedDay === null) issues.push('Weekday is unclear.');
  if (!startTime || !endTime || startTime >= endTime) issues.push('Start/end time is missing or invalid.');
  return {
    subjectName: subjectName.trim(), subjectCode: subjectCode.trim(), weekday: normalizedDay,
    startTime, endTime, location: location.trim(), teacher: teacher.trim(), sessionType: sessionType.trim(), section: section.trim(), sourceLine, issues,
  };
}

const HEADER_MAP = {
  subjectName: ['subject', 'course', 'class name', 'subject name'],
  subjectCode: ['code', 'subject code', 'course code'],
  day: ['day', 'weekday', 'week day'],
  start: ['start', 'start time', 'from'],
  end: ['end', 'end time', 'to', 'until'],
  location: ['room', 'location', 'classroom', 'venue'],
  teacher: ['faculty', 'teacher', 'instructor', 'lecturer'],
};

function parseDelimited(text) {
  const delimiter = text.split(/\r?\n/, 1)[0].includes('\t') ? '\t' : ',';
  const table = parseCsv(text, delimiter);
  if (!table.length) return [];
  const header = table[0].map((value) => value.toLowerCase().replace(/[_-]/g, ' ').trim());
  const positions = Object.fromEntries(Object.entries(HEADER_MAP).map(([field, aliases]) => [field, header.findIndex((item) => aliases.includes(item))]));
  const hasHeader = Object.values(positions).some((index) => index >= 0);
  const rows = hasHeader ? table.slice(1) : table;
  const defaults = { day: 0, start: 1, end: 2, subjectName: 3, subjectCode: 4, location: 5 };
  return rows.map((columns, index) => {
    const get = (field) => columns[positions[field] >= 0 && hasHeader ? positions[field] : defaults[field]] || '';
    return rowResult({ subjectName: get('subjectName'), subjectCode: get('subjectCode'), day: get('day'), start: get('start'), end: get('end'), location: get('location'), teacher: get('teacher') }, index + (hasHeader ? 2 : 1));
  });
}

function parseTextLines(text) {
  const output = []; let currentDay = null;
  const timeRange = /\b(\d{1,2}(?::[0-5]\d)?\s*(?:am|pm)?)\s*(?:-|–|—|to)\s*(\d{1,2}(?::[0-5]\d)?\s*(?:am|pm)?)\b/i;
  text.split(/\r?\n/).forEach((line, index) => {
    const dayMatch = line.match(/\b(Mon(?:day)?|Tues?(?:day)?|Wed(?:nesday)?|Thurs?(?:day)?|Fri(?:day)?|Sat(?:urday)?|Sun(?:day)?)\b/i);
    const dayText = dayMatch?.[0] || (line.match(/[a-z0-9|!]+/gi) || []).find((token) => weekday(token) !== null) || '';
    if (!timeRange.test(line)) {
      if (dayText && line.trim().split(/\s+/).length <= 3) currentDay = dayText;
      return;
    }
    const range = line.match(timeRange); const day = dayText || currentDay;
    const before = line.slice(0, range.index).trim();
    let remainder = `${before} ${line.slice(range.index + range[0].length)}`.replace(/[|,;\t]/g, ' ').trim();
    if (dayText) remainder = remainder.replace(dayText, '').trim();
    remainder = remainder.replace(/^[:\-–—\s]+|[:\-–—\s]+$/g, '');
    if (!day && !remainder) return;
    const tokens = remainder.split(/\s{2,}|\s\|\s/).filter(Boolean);
    const subjectCode = remainder.match(/\b[A-Z]{2,}[A-Z0-9-]*\d[A-Z0-9-]*\b/)?.[0] || '';
    let location = '';
    const room = remainder.match(/\b(?:room|rm|lab|classroom)\s*[:#-]?\s*[A-Z0-9-]+/i);
    if (room) location = room[0];
    else if (tokens.length > 1) location = tokens[tokens.length - 1];
    let subjectName = remainder;
    if (subjectCode) subjectName = subjectName.replace(subjectCode, '').trim();
    if (location && subjectName.endsWith(location)) subjectName = subjectName.slice(0, -location.length).trim();
    output.push(rowResult({ subjectName: subjectName.replace(/^[|,;\s]+|[|,;\s]+$/g, ''), subjectCode, day, start: range[1], end: range[2], location }, index + 1));
  });
  return output;
}

function positionedClock(word) {
  const text = String(word.text || '');
  const match = text.match(/^[-\u2010-\u2015\u2212]?\s*(\d{1,2})[:.](\d{2})\s*(am|pm)?$/i)
    || text.match(/^(am|pm)\s*(\d{1,2})[:.](\d{2})$/i);
  if (!match) return null;
  const prefixed = /^(?:am|pm)/i.test(text);
  return {
    hour: prefixed ? match[2] : match[1],
    minute: prefixed ? match[3] : match[2],
    meridiem: (prefixed ? match[1] : match[3])?.toLowerCase() || '',
    word,
  };
}

function positionedTimeHeaders(lines, dayRows = []) {
  const headers = [];
  for (const line of lines) {
    const words = [...line.words].sort((a, b) => a.bbox.x0 - b.bbox.x0);
    const clocks = words.map((word, index) => ({ ...positionedClock(word), index })).filter((clock) => clock.word);
    for (let index = 0; index < clocks.length - 1; index += 1) {
      const start = clocks[index]; const end = clocks[index + 1];
      const between = words.slice(start.index + 1, end.index);
      const hasSeparator = between.some((word) => /^(?:-|[\u2010-\u2015\u2212]|to)$/i.test(String(word.text || '').trim())) || /^[-\u2010-\u2015\u2212]/.test(String(end.word.text || '').trim());
      if (!hasSeparator) continue;
      const distance = Math.abs((end.word.bbox.x0 + end.word.bbox.x1 - start.word.bbox.x0 - start.word.bbox.x1) / 2);
      const containsUnparsedDigits = between.some((word) => /\d/.test(String(word.text || '')) && !/^(?:\d{1,2}[:.]\d{2})$/i.test(String(word.text || '').trim()));
      if (distance > 200 || containsUnparsedDigits) continue;
      const attachMeridiem = (clock) => {
        if (clock.meridiem) return clock.meridiem;
        const x = (clock.word.bbox.x0 + clock.word.bbox.x1) / 2;
        const adjacent = words.filter((word) => /^(?:am|pm)$/i.test(String(word.text || '').trim()))
          .map((word) => ({ word, distance: Math.abs((word.bbox.x0 + word.bbox.x1) / 2 - x) }))
          .filter((item) => item.distance <= Math.max(72, (clock.word.bbox.y1 - clock.word.bbox.y0) * 2.5))
          .sort((a, b) => a.distance - b.distance)[0];
        return adjacent?.word.text.toLowerCase() || '';
      };
      const startSuffix = attachMeridiem(start); const endSuffix = attachMeridiem(end);
      const startTime = `${start.hour}:${start.minute}${startSuffix ? ` ${startSuffix}` : ''}`;
      const endTime = `${end.hour}:${end.minute}${endSuffix ? ` ${endSuffix}` : ''}`;
      const x = (((start.word.bbox.x0 + start.word.bbox.x1) / 2) + ((end.word.bbox.x0 + end.word.bbox.x1) / 2)) / 2;
      headers.push({ x, left: start.word.bbox.x0, right: end.word.bbox.x1, start: startTime, end: endTime });
    }
  }
  const clocks = lines.flatMap((line, lineIndex) => line.words.map((word) => {
    const clock = positionedClock(word);
    return clock && {
      ...clock,
      lineIndex,
      x: (word.bbox.x0 + word.bbox.x1) / 2,
      y: (word.bbox.y0 + word.bbox.y1) / 2,
      height: word.bbox.y1 - word.bbox.y0,
    };
  }).filter(Boolean));
  const firstDayY = dayRows.length ? Math.min(...dayRows.map((item) => item.y)) : Infinity;
  const headerClocks = clocks.filter((clock) => clock.y < firstDayY);
  for (const start of headerClocks) {
    const end = headerClocks
      .filter((clock) => clock.y > start.y && Math.abs(clock.x - start.x) <= Math.max(24, Math.min(clock.height, start.height) * 1.5))
      .sort((left, right) => left.y - right.y)[0];
    if (!end || end.y - start.y > Math.max(72, Math.max(start.height, end.height) * 4)) continue;
    const suffix = (clock) => {
      if (clock.meridiem) return clock.meridiem;
      const adjacent = lines.flatMap((line) => line.words)
        .filter((word) => /^(?:am|pm)$/i.test(String(word.text || '').trim()))
        .map((word) => ({
          word,
          distance: Math.hypot(
            (word.bbox.x0 + word.bbox.x1) / 2 - clock.x,
            (word.bbox.y0 + word.bbox.y1) / 2 - clock.y
          ),
        }))
        .filter((item) => item.distance <= Math.max(24, clock.height * 1.5))
        .sort((left, right) => left.distance - right.distance)[0];
      return adjacent?.word.text.toLowerCase() || '';
    };
    const startTime = normalizeTime(`${start.hour}:${start.minute}${suffix(start) ? ` ${suffix(start)}` : ''}`);
    const endTime = normalizeTime(`${end.hour}:${end.minute}${suffix(end) ? ` ${suffix(end)}` : ''}`);
    if (!startTime || !endTime || startTime >= endTime) continue;
    const x = (start.x + end.x) / 2;
    if (headers.some((header) => Math.abs(header.x - x) < 18)) continue;
    headers.push({ x, left: Math.min(start.word.bbox.x0, end.word.bbox.x0), right: Math.max(start.word.bbox.x1, end.word.bbox.x1), start: startTime, end: endTime });
  }
  return headers;
}

function extractFacultyDetails(lines) {
  const headerIndex = lines.findIndex((line) => {
    const words = line.words.map((word) => String(word.text || '').toLowerCase());
    return words.some((word) => /^(faculty|instructor|lecturer|teacher)$/i.test(word))
      && words.some((word) => /^code$/i.test(word))
      && words.some((word) => /^subject$/i.test(word));
  });
  if (headerIndex < 0) return new Map();
  const header = lines[headerIndex];
  const centerX = (word) => (word.bbox.x0 + word.bbox.x1) / 2;
  const codeHeader = header.words.find((word) => /^code$/i.test(word.text));
  const subjectHeader = header.words.find((word) => /^subject$/i.test(word.text));
  const facultyHeader = header.words.find((word) => /^(faculty|instructor|lecturer|teacher)$/i.test(word.text));
  const codeX = centerX(codeHeader);
  const subjectX = centerX(subjectHeader);
  const facultyX = centerX(facultyHeader);
  const facultyBoundary = subjectX + (facultyX - subjectX) * 0.68;
  const details = new Map();
  for (const line of lines.slice(headerIndex + 1)) {
    const words = [...line.words].sort((a, b) => a.bbox.x0 - b.bbox.x0);
    const code = words.find((word) => centerX(word) < subjectX - 20 && centerX(word) >= codeX - 30 && /^[A-Z][A-Z0-9-]{1,14}$/i.test(word.text));
    if (!code) continue;
    const subjectName = words
      .filter((word) => centerX(word) >= subjectX - 20 && centerX(word) < facultyBoundary)
      .map((word) => word.text)
      .join(' ')
      .trim();
    const teacher = words
      .filter((word) => centerX(word) >= facultyBoundary)
      .map((word) => word.text)
      .join(' ')
      .trim();
    if (subjectName && teacher) details.set(code.text.toLowerCase(), { code: code.text, subjectName, teacher });
  }
  return details;
}

function parsePositionedWords(words) {
  const ordered = [...words].sort((a, b) => (a.bbox.y0 + a.bbox.y1) / 2 - (b.bbox.y0 + b.bbox.y1) / 2 || a.bbox.x0 - b.bbox.x0);
  const lines = [];
  for (const word of ordered) {
    const midY = (word.bbox.y0 + word.bbox.y1) / 2;
    let line = lines.find((candidate) => Math.abs(candidate.midY - midY) <= Math.max(10, (word.bbox.y1 - word.bbox.y0) * 0.55));
    if (!line) { line = { midY, words: [] }; lines.push(line); }
    line.words.push(word);
    line.midY = line.words.reduce((sum, item) => sum + (item.bbox.y0 + item.bbox.y1) / 2, 0) / line.words.length;
  }
  lines.sort((a, b) => a.midY - b.midY);
  const dayRows = lines.flatMap((line, index) => {
    line.words.sort((a, b) => a.bbox.x0 - b.bbox.x0);
    const anchors = line.words.flatMap((dayWord, wordIndex) => {
      const directDay = weekday(dayWord.text);
      if (directDay !== null) return [{ day: directDay, dayWord }];
      const fragments = [];
      let candidate = '';
      let end = wordIndex - 1;
      for (let offset = wordIndex; offset < Math.min(line.words.length, wordIndex + 7); offset += 1) {
        const current = line.words[offset];
        const fragment = String(current.text || '').replace(/[^a-z0-9]/gi, '');
        if (!fragment || fragment.length > 2 || (end >= wordIndex && current.bbox.x0 - line.words[end].bbox.x1 > Math.max(16, (current.bbox.y1 - current.bbox.y0) * 1.25))) break;
        fragments.push(current);
        candidate += fragment;
        end = offset;
        const day = weekday(candidate);
        if (day !== null && fragments.length >= 2) {
          const joinedWord = {
            text: candidate,
            confidence: fragments.reduce((sum, item) => sum + (Number(item.confidence) || 0), 0) / fragments.length,
            bbox: { x0: fragments[0].bbox.x0, y0: Math.min(...fragments.map((item) => item.bbox.y0)), x1: fragments.at(-1).bbox.x1, y1: Math.max(...fragments.map((item) => item.bbox.y1)) },
            parts: fragments,
          };
          return [{ day, dayWord: joinedWord }];
        }
      }
      return [];
    });
    return anchors.map(({ day, dayWord }) => ({ line, index, day, dayWord, y: (dayWord.bbox.y0 + dayWord.bbox.y1) / 2 }));
  }).sort((a, b) => a.y - b.y);
  if (!dayRows.length) return [];
  const weekdayWords = new Set(dayRows.flatMap(({ dayWord }) => [dayWord, ...(dayWord.parts || [])]));
  const timePattern = /^(\d{1,2}(?::\d{2})?\s*(?:am|pm)?|-\d{1,2}(?::\d{2})?\s*(?:am|pm)?)$/i;
  const clusters = (line, gapFactor = 1.8) => {
    const result = [];
    for (const word of line.words) {
      const last = result[result.length - 1];
      const height = Math.max(10, word.bbox.y1 - word.bbox.y0);
      if (!last || word.bbox.x0 - last.right > Math.max(18, height * gapFactor)) result.push({ words: [word], left: word.bbox.x0, right: word.bbox.x1 });
      else { last.words.push(word); last.right = word.bbox.x1; }
    }
    return result.map((group) => ({ ...group, x: (group.left + group.right) / 2, text: group.words.map((word) => word.text).join(' ') }));
  };
  const rangePattern = /(\d{1,2}(?::\d{2})?\s*(?:am|pm)?)\s*(?:[-\u2010-\u2015\u2212]|to)\s*(\d{1,2}(?::\d{2})?\s*(?:am|pm)?)/i;
  // OCR commonly splits a header such as "09:00 - 10:00" into separate
  // words or even separate lines. Keep the word coordinates and look for a
  // time range within a visual column instead of requiring one OCR line.
  const headers = positionedTimeHeaders(lines, dayRows);
  for (const line of lines) {
    const groups = clusters(line, 2.4);
    for (let start = 0; start < groups.length; start += 1) {
      for (let size = 1; size <= 5 && start + size <= groups.length; size += 1) {
        const group = groups.slice(start, start + size);
        const joined = group.map((part) => part.text).join(' ');
        const range = joined.match(rangePattern);
        if (range && !headers.length) headers.push({ x: (group[0].left + group[group.length - 1].right) / 2, left: group[0].left, right: group[group.length - 1].right, start: range[1], end: range[2] });
      }
    }
  }
  const uniqueHeaders = headers.filter((header, index) => !headers.slice(0, index).some((item) => Math.abs(item.x - header.x) < 18 && item.start === header.start && item.end === header.end));
  const dayGaps = dayRows.slice(1).map((item, index) => item.y - dayRows[index].y).filter((gap) => gap > 0).sort((a, b) => a - b);
  const rowGap = dayGaps.length ? dayGaps[Math.floor((dayGaps.length - 1) / 2)] : 48;
  const halfBand = Math.max(12, Math.min(100, (rowGap || 48) / 2));
  const rows = [];
  const breakPeriods = [];
  for (let dayIndex = 0; dayIndex < dayRows.length; dayIndex += 1) {
    const item = dayRows[dayIndex];
    // The weekday label and the subject baselines often differ enough that
    // Tesseract puts them on separate lines. Collect all words in the visual
    // day band; its edges are halfway to adjacent weekday anchors.
    const upper = dayIndex === 0 ? item.y - halfBand : Math.max((dayRows[dayIndex - 1].y + item.y) / 2, item.y - halfBand);
    const lower = dayIndex === dayRows.length - 1 ? item.y + halfBand : Math.min((item.y + dayRows[dayIndex + 1].y) / 2, item.y + halfBand);
    const cells = ordered.filter((word) => {
      const y = (word.bbox.y0 + word.bbox.y1) / 2;
      return y >= upper && y <= lower && !weekdayWords.has(word) && weekday(word.text) === null && !timePattern.test(word.text);
    });
    if (!cells.length) continue;
    const cellLine = { words: cells.sort((a, b) => a.bbox.x0 - b.bbox.x0) };
    clusters(cellLine, 1.25).forEach((group) => {
      const text = group.text.replace(/[|:;,]+$/g, '').trim();
      let range = text.match(rangePattern);
      if (!range && uniqueHeaders.length) {
        const orderedHeaders = [...uniqueHeaders].sort((a, b) => a.x - b.x);
        const headerIndex = orderedHeaders.reduce((best, header, index) => Math.abs(header.x - group.x) < Math.abs(orderedHeaders[best].x - group.x) ? index : best, 0);
        const header = orderedHeaders[headerIndex];
        const distances = orderedHeaders.map((candidate) => Math.abs(candidate.x - group.x)).sort((a, b) => a - b);
        const leftBound = headerIndex === 0 ? header.x - Math.max(90, (orderedHeaders[1]?.x - header.x || 180) / 2) : (orderedHeaders[headerIndex - 1].x + header.x) / 2;
        const rightBound = headerIndex === orderedHeaders.length - 1 ? header.x + Math.max(90, (header.x - orderedHeaders[headerIndex - 1]?.x || 180) / 2) : (header.x + orderedHeaders[headerIndex + 1].x) / 2;
        const neighborGaps = [orderedHeaders[headerIndex - 1] && header.x - orderedHeaders[headerIndex - 1].x, orderedHeaders[headerIndex + 1] && orderedHeaders[headerIndex + 1].x - header.x].filter(Boolean);
        const maxColumnOffset = Math.min(140, Math.max(60, (Math.min(...neighborGaps, 180) * 0.45)));
        if (group.x >= leftBound && group.x <= rightBound
          && distances[0] <= maxColumnOffset
          && (distances.length < 2 || distances[1] - distances[0] >= 30)) range = header;
      }
      let name = text.replace(/\d{1,2}(?::\d{2})?\s*(?:am|pm)?\s*(?:[-\u2010-\u2015\u2212]|to)\s*\d{1,2}(?::\d{2})?\s*(?:am|pm)?/ig, '').trim();
      if (!name || /^[-\u2010-\u2015\u2212|/]+$/.test(name) || /^(day|subject|faculty|teacher|instructor|semester|section|batch)$/i.test(name)) return;
      const isBreakPeriod = /^(break|lunch|recess|free)(?:\s+(?:break|period|time))?$/i.test(name);
      const locationMatch = name.match(/\b(?:room|rm|lab|classroom)\s*[:#-]?\s*[A-Z0-9-]+/i);
      const location = locationMatch?.[0] || '';
      if (location) name = name.replace(locationMatch[0], '').trim();
      const sessionType = /\blab(?:oratory)?\b/i.test(name) ? 'lab' : /\bpractical\b/i.test(name) ? 'practical' : /\btutorial\b/i.test(name) ? 'tutorial' : '';
      const code = name.match(/\b[A-Z]{2,}[A-Z0-9-]*\d[A-Z0-9-]*\b/)?.[0] || '';
      const sectionLine = lines.slice(0, item.index + 1).reverse().find((candidate) => /\b(section|batch|group|semester|year)\s*[:#-]?\s*[A-Z0-9-]+/i.test(candidate.words.map((word) => word.text).join(' ')));
      const section = sectionLine?.words.map((word) => word.text).join(' ').trim() || '';
      let subjectName = code ? name.replace(code, '').trim() || code : name;
      if (sessionType) {
        const withoutSessionType = subjectName.replace(/\b(lab(?:oratory)?|practical|tutorial)\b/ig, '').replace(/\s{2,}/g, ' ').trim();
        if (withoutSessionType) subjectName = withoutSessionType;
      }
      const row = rowResult({ subjectName: isBreakPeriod ? name : subjectName, subjectCode: code, day: item.day, start: range?.start || range?.[1], end: range?.end || range?.[2], location, sessionType, section }, item.index + 1);
      if (isBreakPeriod) row.isBreakPeriod = true;
      const confidenceWords = [...group.words, item.dayWord].filter((word) => Number.isFinite(word.confidence));
      if (confidenceWords.length) {
        row.confidence = Math.round(confidenceWords.reduce((sum, word) => sum + word.confidence, 0) / confidenceWords.length);
        if (row.confidence < 55) row.issues.push('OCR confidence is low. Verify this row against the image.');
      }
      if (isBreakPeriod) breakPeriods.push(row);
      else rows.push(row);
    });
  }
  const facultyDetails = extractFacultyDetails(lines);
  for (const row of rows) {
    const detail = facultyDetails.get((row.subjectCode || row.subjectName).toLowerCase());
    if (!detail) continue;
    row.subjectCode = detail.code;
    if (row.subjectName.toLowerCase() === detail.code.toLowerCase()) row.subjectName = detail.subjectName;
    row.teacher = detail.teacher;
  }
  return [...rows, ...breakPeriods];
}

function parseTimetable(text, sourceType, words = []) {
  if (words.length) {
    const positioned = parsePositionedWords(words);
    if (positioned.length) return positioned.map((row, index) => ({ ...row, sourceLine: row.sourceLine || index + 1 }));
  }
  const delimited = sourceType === 'text/csv' || sourceType === 'text/tab-separated-values' || /,|\t/.test(text.split(/\r?\n/, 1)[0]);
  const rows = delimited ? parseDelimited(text) : parseTextLines(text);
  if (!rows.length) {
    const hasDay = words.some((word) => weekday(word.text) !== null) || hasWeekdayText(text);
    const message = sourceType.startsWith('image/')
      ? (hasDay
        ? 'Weekdays were detected, but the timetable rows and period times could not be associated. Try a clearer, upright photo, or start a manual preview to enter the classes you can read.'
        : 'Image text was detected, but no timetable weekdays or class rows were recognized. Try a clearer photo focused on the timetable grid, or start a manual preview.')
      : 'No timetable rows were recognized. Use a CSV with subject, day, startTime, and endTime columns, or try a clearer text-based timetable.';
    throw Object.assign(new Error(message), { statusCode: 422 });
  }
  return rows.map((row, index) => ({ ...row, sourceLine: row.sourceLine || index + 1 }));
}

function cleanPayload(resource, body) {
  if (!body || typeof body !== 'object' || Array.isArray(body)) throw new AppError(400, 'Request body must be an object.');
  const fields = {
    subjects: ['name', 'code', 'teacher', 'credits', 'description'],
    timetable: ['subject', 'weekday', 'startTime', 'endTime', 'location', 'teacher', 'sessionType'],
    calendar: ['title', 'type', 'date', 'endDate', 'description'],
    exams: ['title', 'type', 'subject', 'date', 'startTime', 'endTime', 'venue', 'notes'],
    syllabus: ['subject', 'unit', 'topic', 'completed', 'progress'],
  }[resource];
  if (!fields) throw new AppError(404, 'Academic resource not found.');
  const payload = {};
  fields.forEach((field) => { if (body[field] !== undefined) payload[field] = body[field]; });
  if (resource === 'syllabus') {
    if (payload.progress !== undefined && !['not-started', 'in-progress', 'completed'].includes(payload.progress)) {
      throw new AppError(400, 'Choose Not Started, In Progress, or Completed topic progress.');
    }
    if (payload.progress !== undefined) payload.completed = payload.progress === 'completed';
    else if (payload.completed !== undefined) payload.progress = payload.completed ? 'completed' : 'not-started';
  }
  for (const key of ['subject']) {
    if (payload[key] !== undefined && !mongoose.isValidObjectId(payload[key])) throw new AppError(400, `Invalid ${key} identifier.`, { [key]: `Invalid ${key} identifier.` });
  }
  return payload;
}

function escapedRegex(value) { return String(value || '').replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }

async function assertNoDuplicate(resource, values, userId, excludedId) {
  let identity;
  if (resource === 'timetable') {
    if (!values.subject || values.weekday == null || !values.startTime || !values.endTime) return;
    identity = { subject: values.subject, weekday: values.weekday, startTime: values.startTime, endTime: values.endTime };
  } else if (resource === 'calendar') {
    if (!values.title || !values.date || Number.isNaN(new Date(values.date).getTime())) return;
    identity = { title: new RegExp(`^${escapedRegex(values.title)}$`, 'i'), date: values.date };
  } else if (resource === 'exams') {
    if (!values.subject || !values.title || !values.date || Number.isNaN(new Date(values.date).getTime())) return;
    identity = { subject: values.subject, title: new RegExp(`^${escapedRegex(values.title)}$`, 'i'), date: values.date, startTime: values.startTime || '' };
  } else if (resource === 'syllabus') {
    if (!values.subject || !values.unit || !values.topic) return;
    identity = { subject: values.subject, unit: values.unit, topic: values.topic };
  }
  else return;
  if (Object.values(identity).some((value) => value === undefined)) return;
  const query = { ...identity, user: userId };
  if (excludedId) query._id = { $ne: excludedId };
  if (await definitions[resource].model.exists(query)) throw new AppError(409, `A matching ${resource === 'timetable' ? 'class' : resource === 'calendar' ? 'calendar event' : resource === 'exams' ? 'exam' : 'syllabus topic'} already exists.`);
}

function controller(resource) {
  const { model, populate, sort = { createdAt: -1 } } = definitions[resource];
  return {
    list: asyncHandler(async (req, res) => {
      const query = model.find({ user: req.user.id }).sort(sort);
      if (populate) query.populate(populate, 'name code');
      const records = await query;
      res.json({ success: true, records });
    }),
    create: asyncHandler(async (req, res) => {
      const values = cleanPayload(resource, req.body);
      await assertSubjectOwner(values.subject, req.user.id);
      await assertNoDuplicate(resource, values, req.user.id);
      const record = await model.create({ ...values, user: req.user.id });
      if (populate) await record.populate(populate);
      res.status(201).json({ success: true, record });
    }),
    update: asyncHandler(async (req, res) => {
      const values = cleanPayload(resource, req.body);
      await assertSubjectOwner(values.subject, req.user.id);
      const record = await model.findOne({ _id: req.params.id, user: req.user.id });
      if (!record) throw new AppError(404, 'Record not found.');
      await assertNoDuplicate(resource, { ...record.toObject(), ...values }, req.user.id, record._id);
      Object.assign(record, values);
      await record.save();
      if (populate) await record.populate(populate);
      res.json({ success: true, record });
    }),
    remove: asyncHandler(async (req, res) => {
      const record = await model.findOneAndDelete({ _id: req.params.id, user: req.user.id });
      if (!record) throw new AppError(404, 'Record not found.');
      if (resource === 'subjects') {
        const entries = await TimetableEntry.find({ user: req.user.id, subject: record.id }).select('_id');
        await Promise.all([
          TimetableEntry.deleteMany({ user: req.user.id, subject: record.id }),
          Exam.deleteMany({ user: req.user.id, subject: record.id }),
          SyllabusTopic.deleteMany({ user: req.user.id, subject: record.id }),
          Attendance.deleteMany({ user: req.user.id, subject: record.id }),
          ClassIntention.deleteMany({ user: req.user.id, timetableEntry: { $in: entries.map((entry) => entry._id) } }),
          StudySession.deleteMany({ user: req.user.id, subject: record.id }),
        ]);
      }
      if (resource === 'timetable') {

          Attendance.updateMany({ user: req.user.id, timetableEntry: record.id }, { $set: { timetableEntry: null } }),
          ClassIntention.deleteMany({ user: req.user.id, timetableEntry: record.id }),
        ]);
      }
      if (resource === 'exams') await StudySession.deleteMany({ user: req.user.id, exam: record.id });
      if (resource === 'syllabus') await StudySession.deleteMany({ user: req.user.id, topic: record.id, status: 'planned' });
      res.json({ success: true });
    }),
  };

function makeTextPdf(text) {
  const escaped = text.replace(/([\\()])/g, '\\$1');
  const stream = `BT /F1 12 Tf 72 720 Td (${escaped}) Tj ET`;
  const objects = [
    '<< /Type /Catalog /Pages 2 0 R >>',
    '<< /Type /Pages /Kids [3 0 R] /Count 1 >>',
    '<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources << /Font << /F1 4 0 R >> >> /Contents 5 0 R >>',
    '<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>',
    `<< /Length ${Buffer.byteLength(stream)} >>\nstream\n${stream}\nendstream`,
  ];
  let pdf = '%PDF-1.4\n';
  const offsets = [0];
  objects.forEach((object, index) => { offsets.push(Buffer.byteLength(pdf)); pdf += `${index + 1} 0 obj\n${object}\nendobj\n`; });
  const xref = Buffer.byteLength(pdf);
  pdf += `xref\n0 ${objects.length + 1}\n0000000000 65535 f \n${offsets.slice(1).map((offset) => `${String(offset).padStart(10, '0')} 00000 n \n`).join('')}trailer\n<< /Size ${objects.length + 1} /Root 1 0 R >>\nstartxref\n${xref}\n%%EOF`;
  return Buffer.from(pdf, 'ascii');
}

const FILE_ACCEPT = '.pdf,.csv,.tsv,.txt,.jpg,.jpeg,.png,.webp,application/pdf,text/csv,text/plain,image/*';
const normalize = (value) => String(value || '').trim().toLowerCase();

function findSubject(row, subjects) {
  const code = normalize(row.subjectCode);
  const name = normalize(row.subjectName);
  const matches = subjects.filter((subject) => (code && normalize(subject.code) === code) || (name && normalize(subject.name) === name));
  return matches.length === 1 ? String(matches[0]._id || matches[0].id) : '';
}

export default function TimetableImport({ subjects, onImported }) {
  const [file, setFile] = useState(null);
  const [rows, setRows] = useState([]);
  const [method, setMethod] = useState('');
  const [selectedSection, setSelectedSection] = useState('');
  const [notice, setNotice] = useState('');
  const [error, setError] = useState('');
  const [parsing, setParsing] = useState(false);
  const [saving, setSaving] = useState(false);

  async function preview(event) {
    event.preventDefault();
    if (!file) { setError('Choose a timetable file first.'); return; }
    if (file.size > 12 * 1024 * 1024) { setError('Files must be 12 MB or smaller.'); return; }
    setError(''); setNotice(''); setRows([]); setParsing(true);
    try {
      const result = await documentImportApi.previewTimetable(file);
      setMethod(result.extraction?.method || 'text extraction');
      const extractedRows = result.rows || [];
      const sections = [...new Set(extractedRows.map((row) => row.section).filter(Boolean))];
      setSelectedSection(sections.length > 1 ? sections[0] : '');
      setRows(extractedRows.map((row, index) => ({ ...row, key: `${Date.now()}-${index}`, subjectId: findSubject(row, subjects), saved: false })));
      setNotice(result.message || 'Review every row before saving.');
    } catch (err) { setError(err.message); }
    finally { setParsing(false); }
  }

  function changeRow(key, field, value) {
    setRows((current) => current.map((row) => {
      if (row.key !== key) return row;
      const next = { ...row, [field]: value };
      const issues = [];
      const selected = subjects.find((subject) => String(subject._id || subject.id) === String(next.subjectId));
      if (!next.subjectName.trim() && !selected) issues.push('Subject is required.');
      if (next.weekday === null || next.weekday === '') issues.push('Choose a weekday.');
      if (!next.startTime || !next.endTime || next.startTime >= next.endTime) issues.push('Use a valid time range.');
      return { ...next, issues };
    }));
  }

  function removeRow(key) { setRows((current) => current.filter((row) => row.key !== key)); }

  function addRow() {
    setRows((current) => [...current, { key: `manual-${Date.now()}`, subjectName: '', subjectCode: '', subjectId: '', weekday: null, startTime: '', endTime: '', location: '', teacher: '', sessionType: '', section: selectedSection, sourceLine: 'manual', issues: ['Complete this entry before saving.'], saved: false }]);
  }

    const pending = visibleRows.filter((row) => !row.saved);
    if (!pending.length) return;
    const invalid = pending.find((row) => row.issues.length || !row.subjectId);

      setError(!invalid.subjectId ? `Select a subject for “${invalid.subjectName || 'unnamed row'}” before saving.` : `Correct row ${invalid.sourceLine} before saving.`);
      return;

        setRows((current) => current.map((item) => item.key === row.key ? { ...item, saved: true } : item));
        count += 1;
      } catch (err) { failures.push(`Row ${row.sourceLine}: ${err.message}`); }
    }
    await onImported();
    if (failures.length) {
      setError(`${count} row(s) saved. ${failures.join(' ')}`);
      setNotice('You can correct or retry the rows that did not save. Already-saved rows will not be submitted again.');
    } else {
      setNotice(`${count} timetable row(s) saved and linked to their subjects.`);
    }
    setSaving(false);
  }

  const breakPeriods = rows.filter((row) => row.isBreakPeriod);
  const visibleRows = rows.filter((row) => !row.isBreakPeriod && (!selectedSection || !row.section || row.section === selectedSection));
  const pendingRows = visibleRows.filter((row) => !row.saved);

  return <section className="timetable-import card" aria-labelledby="timetable-import-title">

These will not be saved as classes.</div>}

// Framework-agnostic API client. It only uses fetch, so the same module can be
// reused by the React Native app later (which would pass a Bearer token instead
// of relying on the browser's httpOnly cookie).
const BASE_URL = (import.meta.env.VITE_API_URL || 'http://localhost:5000/api').replace(/\/$/, '');

export class ApiError extends Error {
  constructor(message, { status = 0, errors = {} } = {}) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
    this.errors = errors; // field-level messages, e.g. { email: '...' }
  }
}

async function request(path, { method = 'GET', body } = {}) {
  let response;
  try {
    response = await fetch(`${BASE_URL}${path}`, {
      method,
      // Sends and receives the httpOnly auth cookie.
      credentials: 'include',
      headers: body ? { 'Content-Type': 'application/json' } : undefined,
      body: body ? JSON.stringify(body) : undefined,
    });
  } catch {
    throw new ApiError('Cannot reach the Acadova server. Check your connection and try again.');
  }

  let data = null;
  try {
    data = await response.json();
  } catch {
    // Non-JSON response; handled below.
  }

  if (!response.ok) {
    throw new ApiError(data?.message || 'Something went wrong. Please try again.', {
      status: response.status,
      errors: data?.errors,
    });
  }
  return data;
}

export const api = {
  get: (path) => request(path),
  post: (path, body) => request(path, { method: 'POST', body: body ?? {} }),
  put: (path, body) => request(path, { method: 'PUT', body: body ?? {} }),
  delete: (path) => request(path, { method: 'DELETE' }),
};
